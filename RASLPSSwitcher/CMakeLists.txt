cmake_minimum_required(VERSION 3.16.3...3.28.2 FATAL_ERROR)
project(RASLPSSwitcher)
set(EXTENSION_HOMEPAGE "https://www.slicer.org/wiki/Documentation/Nightly/Extensions/RASLPSSwitcher")
set(EXTENSION_CONTRIBUTORS "Pedro (Developer)")
set(EXTENSION_DESCRIPTION "Alterna entre RAS e LPS com 1 clique para modelos e segmentacoes (inverte X e Y). Operacao reversivel e aplicavel a multiplos nodes.")
set(EXTENSION_ICONURL "")
set(EXTENSION_SCREENSHOTURLS "")
set(EXTENSION_DEPENDS "NA")
find_package(Slicer REQUIRED)
include(${Slicer_USE_FILE})
add_subdirectory(RASLPSSwitcher)
include(${Slicer_EXTENSION_GENERATE_CONFIG})
include(${Slicer_EXTENSION_CPACK})
